    <script>
        // Web Audio Synthesizer
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx = null;
        let isMuted = false;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
        }

        function playPopSound() {
            if (isMuted) return;
            initAudio();
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(800, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.08);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.08);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.08);
        }

        function playChime() {
            if (isMuted) return;
            initAudio();
            const notes = [523.25, 659.25, 783.99, 1046.50];
            notes.forEach((freq, idx) => {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(freq, audioCtx.currentTime + idx * 0.1);
                gain.gain.setValueAtTime(0.15, audioCtx.currentTime + idx * 0.1);
                gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + idx * 0.1 + 0.6);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start(audioCtx.currentTime + idx * 0.1);
                osc.stop(audioCtx.currentTime + idx * 0.1 + 0.6);
            });
        }

        function toggleAudio() {
            isMuted = !isMuted;
            document.getElementById('audioToggle').innerText = isMuted ? '🔇' : '🎵';
        }

        // Smooth Page Flow System
        let currentPage = 1;

        function goToPage(pageNum) {
            initAudio();
            playChime();

            const currentEl = document.getElementById(`page${currentPage}`);
            const nextEl = document.getElementById(`page${pageNum}`);

            currentEl.classList.remove('active');
            currentEl.classList.add('exit');

            setTimeout(() => {
                currentEl.classList.remove('exit');
                nextEl.classList.add('active');
                currentPage = pageNum;

                if (pageNum === 3) {
                    triggerCelebration();
                }
            }, 400);
        }

        // Particle Canvas Animation
        const canvas = document.getElementById('bgCanvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        class Particle {
            constructor() {
                this.reset();
            }
            reset() {
                this.x = Math.random() * canvas.width;
                this.y = Math.random() * canvas.height;
                this.size = Math.random() * 2 + 0.5;
                this.speedX = (Math.random() - 0.5) * 0.5;
                this.speedY = -Math.random() * 0.6 - 0.2;
                this.alpha = Math.random() * 0.7 + 0.3;
            }
            update() {
                this.x += this.speedX;
                this.y += this.speedY;
                if (this.y < 0) this.reset();
            }
            draw() {
                ctx.fillStyle = `rgba(255, 183, 197, ${this.alpha})`;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        for (let i = 0; i < 60; i++) {
            particles.push(new Particle());
        }

        function animateParticles() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            particles.forEach(p => {
                p.update();
                p.draw();
            });
            requestAnimationFrame(animateParticles);
        }
        animateParticles();

        // Celebration & Confetti Engine
        function triggerCelebration() {
            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 100,
                    spread: 70,
                    origin: { y: 0.6 },
                    colors: ['#ffb7c5', '#ffd700', '#8a2be2', '#ff416c']
                });
            }
            spawnPetals();
            spawnBalloons();
        }

        // Floating Rose Petals
        function spawnPetals() {
            for (let i = 0; i < 15; i++) {
                setTimeout(() => {
                    const petal = document.createElement('div');
                    petal.className = 'petal';
                    petal.style.left = Math.random() * 100 + 'vw';
                    petal.style.width = (Math.random() * 10 + 10) + 'px';
                    petal.style.height = (Math.random() * 10 + 10) + 'px';
                    petal.style.animationDuration = (Math.random() * 3 + 4) + 's';
                    document.body.appendChild(petal);

                    setTimeout(() => petal.remove(), 7000);
                }, i * 300);
            }
        }

        // Interactive Balloons
        function spawnBalloons() {
            const container = document.getElementById('balloonContainer');
            const colors = ['#ff416c', '#8a2be2', '#ffd700', '#ffb7c5', '#00d2ff'];

            for (let i = 0; i < 8; i++) {
                setTimeout(() => {
                    const b = document.createElement('div');
                    b.className = 'balloon';
                    b.style.left = (Math.random() * 85 + 5) + 'vw';
                    b.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
                    b.style.animationDuration = (Math.random() * 4 + 6) + 's';

                    b.onclick = function() {
                        playPopSound();
                        if (typeof confetti === 'function') {
                            confetti({
                                particleCount: 25,
                                spread: 50,
                                origin: {
                                    x: b.getBoundingClientRect().left / window.innerWidth,
                                    y: b.getBoundingClientRect().top / window.innerHeight
                                }
                            });
                        }
                        b.remove();
                    };

                    container.appendChild(b);
                }, i * 600);
            }
        }

        // Cake Candle Blow Feature
        let isCandleBlown = false;
        function blowCandle() {
            if (isCandleBlown) return;
            isCandleBlown = true;

            const flame = document.getElementById('candleFlame');
            const smoke = document.getElementById('candleSmoke');

            flame.classList.add('extinguished');
            smoke.classList.add('active');

            playPopSound();
            if (typeof confetti === 'function') {
                confetti({ particleCount: 80, spread: 80, origin: { y: 0.5 } });
            }
        }

        // Gift Unwrapping

            const card = document.getElementById('wishCard');
            card.style.display = 'block';
            playChime();
            if (typeof confetti === 'function') {
                confetti({ particleCount: 50, spread: 60, origin: { y: 0.7 } });
            }
        }
